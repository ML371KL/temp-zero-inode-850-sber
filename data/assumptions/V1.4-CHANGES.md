```

# stage1/calib/nii/calib_nii.py (калибровка книг ЧПД на 2026Q3)
python -B ops/tools/build_facts.py --anchor 2026Q3 --out var/reanchor/facts-2026Q3 --report var/reanchor/report-2026Q3.json
python -B ops/tools/reanchor.py --facts var/reanchor/report-2026Q3.json --out var/reanchor/2026Q3 --valuation-date 2026-10-28 \
       --band 2000 --compare-facts var/reanchor/facts-2026Q3 --observation var/reanchor/observation-2026Q3.yaml
python -B ops/tools/build_facts.py --anchor 2026Q3 && python -B ops/tools/build_facts.py --check
python -B ops/tools/build_facts.py --check --sheet-template data/facts/sheets/2026Q4.csv
python -B data/assumptions/build_assumptions.py [--check]           # шаблон кандидата + суждения §3 → машинная книга 1.4
python -B data/assumptions/evidence/nii/sigma0_split.py && python -B data/assumptions/evidence/nii/phi_split.py
python -B -m model.book_results && python -B ops/tools/render_numbers.py && python -B -m tests.test_control_model --write
python -B data/assumptions/evidence/governance/governance_channels.py   # вход median_rub — медиана книги 1.4
python -B ops/tools/walk_book.py --start var/reanchor/2026Q3/assumptions.json --end data/assumptions/assumptions.json \
       --steps var/reanchor/walk-1.4-steps.yaml --median
```
